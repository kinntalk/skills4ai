**误报示例**:

```python
               mode='r',
```

**影响**: 中等 - 可能导致开发者收到不必要的警告

**改进建议**:
```python
# 改进方案：使用 AST 解析而非简单的行匹配
import ast

def check_encoding_safety(skill_path):
    issues = []
    for py_file in skill_path.glob('**/*.py'):
        try:
            content = py_file.read_text(encoding='utf-8')
            tree = ast.parse(content)
            
            for node in ast.walk(tree):
                # 检查 open() 调用
                if isinstance(node, ast.Call):
                    if isinstance(node.func, ast.Name) and node.func.id == 'open':
                        # 检查是否有 encoding 参数
                        has_encoding = any(
                            kw.arg == 'encoding' for kw in node.keywords
                        )
                        if not has_encoding:
                            # 检查模式是否为二进制
                            mode = None
                            for kw in node.keywords:
                                if kw.arg == 'mode':
                                    if isinstance(kw.value, ast.Constant):
                                        mode = kw.value.value
                            if mode and 'b' in mode:
                                continue
                            issues.append(f"{py_file.name}:{node.lineno}: open() call without explicit encoding")
        except Exception as e:
            issues.append(f"Could not parse {py_file.name}: {e}")
    
    return (False, issues) if issues else (True, "File operations appear to use explicit encoding")
```

```python
def check_subprocess_robustness(skill_path):
    issues = []
    
    for py_file in skill_path.glob('**/*.py'):
        if py_file.name == 'audit_skill.py':
            continue
        try:
            content = py_file.read_text(encoding='utf-8')
            tree = ast.parse(content)
            
            for node in ast.walk(tree):
                if isinstance(node, ast.Call):
                    if isinstance(node.func, ast.Attribute):
                        if node.func.attr in ['run', 'check_output', 'call', 'Popen']:
                            # 检查 shell=True
                            for kw in node.keywords:
                                if kw.arg == 'shell' and isinstance(kw.value, ast.Constant):
                                    if kw.value.value is True:
                                        issues.append(f"{py_file.name}:{node.lineno}: subprocess.{node.func.attr}() with shell=True is a security risk")
                            
                            # 检查未验证的用户输入
                            if node.args:
                                for arg in node.args:
                                    if isinstance(arg, ast.Name):
                                        # 简单检查：如果参数是变量名，可能需要验证
                                        issues.append(f"{py_file.name}:{node.lineno}: subprocess.{node.func.attr}() uses variable '{arg.id}' without validation (potential security risk)")
                            
                            # 检查 encoding 处理
                            if 'text=True' in ast.unparse(node) or 'encoding=' in ast.unparse(node):
                                if 'errors=' not in ast.unparse(node):
                                    issues.append(f"{py_file.name}:{node.lineno}: Subprocess call might crash on non-UTF8 output (missing errors='replace' or similar)")
        except Exception as e:
            issues.append(f"Could not parse {py_file.name}: {e}")
    
    return (False, issues) if issues else (True, "Subprocess calls appear robust or binary")
```

```python
def check_absolute_references(skill_path):
    issues = []
    
    for py_file in skill_path.glob('**/*.py'):
        try:
            content = py_file.read_text(encoding='utf-8')
            lines = content.splitlines()
            
            for i, line in enumerate(lines, 1):
                stripped = line.strip()
                
                # 跳过注释和导入
                if stripped.startswith('#') or stripped.startswith(('import', 'from')):
```

```python
def check_dependencies(skill_path):
    """检查依赖完整性（双向检查）"""
    scripts_dir = skill_path / 'scripts'
    if not scripts_dir.exists():
        return True, "No scripts directory"
    
    py_files = list(scripts_dir.glob('**/*.py'))
    if not py_files:
        return True, "No Python scripts found"
        
    req_file = scripts_dir / 'requirements.txt'
    if not req_file.exists():
        return False, "Python scripts found but scripts/requirements.txt is missing"
    
    # 扫描代码中使用的模块
    imported_modules = set()
    std_lib = sys.stdlib_module_names if hasattr(sys, 'stdlib_module_names') else set()
    
    if not std_lib:
         std_lib = {'os', 'sys', 're', 'json', 'yaml', 'pathlib', 'argparse', 'subprocess', 'shutil', 'tempfile', 'time', 'datetime', 'logging', 'threading', 'typing', 'collections', 'io', 'math', 'random', 'string', 'hashlib', 'base64', 'urllib', 'http', 'email', 'csv', 'sqlite3', 'configparser', 'zipfile', 'tarfile', 'gzip', 'bz2', 'pickle', 'copy', 'itertools', 'functools', 'operator', 'decimal', 'fractions', 'statistics', 'enum', 'dataclasses', 'uuid', 'secrets', 'inspect', 'warnings', 'contextlib', 'abc', 'numbers', 'types'}

    for py_file in py_files:
        try:
            content = py_file.read_text(encoding='utf-8')
            imports = re.findall(r'^\s*(?:import|from)\s+([a-zA-Z0-9_]+)', content, re.MULTILINE)
            for module in imports:
                if module not in std_lib and module != 'scripts':
                    imported_modules.add(module)
        except Exception as e:
            print(f"Warning: Could not decode {py_file.name}: {e}")

    # 读取 requirements.txt
    try:
        req_content = req_file.read_text(encoding='utf-8').lower()
        declared_deps = set(line.split('==')[0].split('>=')[0].split('~=')[0].strip() for line in req_content.splitlines() if line.strip() and not line.startswith('#'))
    except Exception:
        return False, "Could not read requirements.txt"

    # 包名映射
    pkg_map = {
        'yaml': 'pyyaml',
        'PIL': 'pillow',
        'bs4': 'beautifulsoup4',
        'dotenv': 'python-dotenv',
        'git': 'gitpython',
        'cv2': 'opencv-python',
        'sklearn': 'scikit-learn',
    }

    issues = []
    
    # 检查缺少的依赖
    missing_deps = []
    for module in imported_modules:
        pkg_name = pkg_map.get(module, module).lower()
        if pkg_name not in declared_deps and module.lower() not in declared_deps:
            if not (scripts_dir / f"{module}.py").exists():
                 missing_deps.append(f"{module} (package: {pkg_name})")

    if missing_deps:
        issues.append(f"Potential missing dependencies in requirements.txt: {', '.join(missing_deps)}")
    
    # 检查未使用的依赖
    unused_deps = []
    for dep in declared_deps:
        # 检查是否在代码中使用
        used = False
        for module in imported_modules:
            pkg_name = pkg_map.get(module, module).lower()
            if dep == pkg_name or dep == module.lower():
                used = True
                break
        if not used:
            unused_deps.append(dep)
    
    if unused_deps:
        issues.append(f"Potentially unused dependencies in requirements.txt: {', '.join(unused_deps)} (consider removing)")

    if issues:
        return False, issues

    return True, "Dependency configuration looks good"
```

```python
    ref_doc_patterns = [
        r'.*-tracing\.md$',
        r'.*-guide\.md$',
        r'.*-protocol\.md$',
        r'.*-reference\.md$',
        r'.*-workflow\.md$',
        r'.*-methodology\.md$'
    ]
    unexpected_files = []
    
    try:
        for item in skill_path.iterdir():
            if item.is_file():
                if item.name not in allowed_files:
                    is_ref_doc = any(re.match(pattern, item.name) for pattern in ref_doc_patterns)
                    if not is_ref_doc:
                        unexpected_files.append(item.name)
    except Exception as e:
        return False, f"Could not scan directory: {e}"
    
    issues = []
    if unexpected_files:
        issues.append(f"Unexpected top-level files: {', '.join(unexpected_files)}")
    
    if issues:
        return False, issues
    return True, f"Directory structure is valid (found: {', '.join(found_dirs) if found_dirs else 'none'})"
```

```python
def check_i18n_support(skill_path):
    """
    Check for internationalization (i18n) and multi-language support.
    """
    issues = []
    
    # 检查 SKILL.md 的多语言支持
    skill_md = skill_path / 'SKILL.md'
    if skill_md.exists():
        try:
            content = skill_md.read_text(encoding='utf-8')
            
            # 更精确的多语言检查
            has_english_keywords = any(word in content.lower() for word in ['description:', 'name:', 'usage:', 'example', 'overview'])
            has_chinese_keywords = any(word in content for word in ['描述', '名称', '用法', '示例', '概述'])
            has_chinese_chars = any(ord(char) > 127 for char in content)
            
            # 检查是否有 description_zh 字段
            has_description_zh = 'description_zh:' in content
            
            if not has_description_zh and not has_chinese_keywords:
                if has_english_keywords:
                    issues.append("Suggestion: Consider adding Chinese translation (description_zh) in SKILL.md for better discoverability.")
            
            if not has_english_keywords and has_chinese_chars:
                issues.append("Suggestion: Consider adding English keywords in SKILL.md for better discoverability.")
                
        except Exception as e:
            issues.append(f"Could not read SKILL.md: {e}")
    
    # 检查 Python 文件中的硬编码输出消息
    for py_file in skill_path.glob('**/*.py'):
        try:
            content = py_file.read_text(encoding='utf-8')
            lines = content.splitlines()
            
            message_count = 0
            for i, line in enumerate(lines, 1):
                stripped = line.strip()
                
                # 跳过注释
                if stripped.startswith('#'):
                    continue
                
                # 统计 print 语句中的硬编码字符串
                if 'print("' in line or "print('" in line:
                    message_count += 1
                
                # 检查 emoji 使用（严格：不允许在 skill 代码输出中使用 emoji）
                if 'print(' in line:
                    has_emoji = False
                    for c in line:
                        if (0x2600 <= ord(c) <= 0x27BF) or (0x1F300 <= ord(c) <= 0x1F9FF):
                            has_emoji = True
                            break
                    
                    if has_emoji:
                        # 允许注释中的 Unicode
                        if '#' in line:
                            comment_part = line.split('#', 1)[1]
                            emoji_in_comment = False
                            for c in comment_part:
                                if (0x2600 <= ord(c) <= 0x27BF) or (0x1F300 <= ord(c) <= 0x1F9FF):
                                    emoji_in_comment = True
                                    break
                            if emoji_in_comment:
                                continue
                        
                        issues.append(f"{py_file.name}:{i}: Emoji found in output statement. Emoji is not allowed in skill code. Use standard text labels [PASS]/[FAIL]/[WARN]/[INFO] instead.")
            
            # 如果有大量硬编码消息，建议使用消息字典
            if message_count > 20:
                issues.append(f"Suggestion: {py_file.name} has {message_count} print statements. Consider using a message dictionary for better i18n support when applicable.")
                    
        except Exception as e:
            issues.append(f"Could not read {py_file.name}: {e}")
            
    if issues:
        # 分离 emoji 错误和其他 i18n 问题
        emoji_issues = [i for i in issues if "Emoji found" in i]
        other_issues = [i for i in issues if "Emoji found" not in i]
        
        if emoji_issues:
            return False, issues
            
        return False, issues
    return True, "Internationalization check completed"
```
